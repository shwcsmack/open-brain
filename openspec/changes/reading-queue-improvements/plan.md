# Reading Queue Improvements — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add archive/delete item management, Wikipedia link tracking with push-to-front navigation, whole-article Wikipedia import, inline passage deletion with tombstones, a stable article detail page, and an archive log page to the reading queue.

**Architecture:** Core logic lives in `lib/readingQueue.ts`; tRPC procedures in `server/routers/reading.ts` wire it to the frontend. UI is entirely client-rendered (`'use client'`) using tRPC hooks. New fields `archivedAt` and `hiddenPassages` are added to `ReadingItem` via a Prisma migration that also hard-deletes all `WIKIPEDIA_SECTION` rows and renames the enum value to `WIKIPEDIA`.

**Tech Stack:** Next.js (App Router, client components), tRPC, Prisma + SQLite (better-sqlite3 adapter), TypeScript, Tailwind CSS, shadcn/ui, Sonner (toasts). Tests: `node:test` + `tsx`.

```typescript
export async function hidePassage(db: PrismaClient, id: string, text: string) {
  const item = await db.readingItem.findUniqueOrThrow({ where: { id } })
  const current = JSON.parse(item.hiddenPassages) as string[]
  return db.readingItem.update({
    where: { id },
    data: { hiddenPassages: JSON.stringify([...current, text]) },
  })
}

export async function restorePassage(db: PrismaClient, id: string, text: string) {
  const item = await db.readingItem.findUniqueOrThrow({ where: { id } })
  const current = JSON.parse(item.hiddenPassages) as string[]
  const idx = current.indexOf(text)
  if (idx === -1) return item
  const next = [...current.slice(0, idx), ...current.slice(idx + 1)]
  return db.readingItem.update({ where: { id }, data: { hiddenPassages: JSON.stringify(next) } })
}
```

```typescript
export async function getItemById(db: PrismaClient, id: string) {
  return db.readingItem.findFirst({ where: { id, deletedAt: null } })
}

export async function listArchivedItems(
  db: PrismaClient,
  filters?: { sourceType?: 'NOTE' | 'URL' | 'WIKIPEDIA' | 'EXTRACT' }
) {
  return db.readingItem.findMany({
    where: {
      deletedAt: null,
      archivedAt: { not: null },
      ...(filters?.sourceType ? { sourceType: filters.sourceType } : {}),
    },
    orderBy: { archivedAt: 'desc' },
  })
}

export async function getImportedWikipediaUrls(
  db: PrismaClient
): Promise<Record<string, { id: string; archivedAt: Date | null }>> {
  const items = await db.readingItem.findMany({
    where: { deletedAt: null, articleUrl: { not: null } },
    select: { id: true, articleUrl: true, archivedAt: true },
  })
  const map: Record<string, { id: string; archivedAt: Date | null }> = {}
  for (const item of items) {
    if (item.articleUrl) map[item.articleUrl] = { id: item.id, archivedAt: item.archivedAt }
  }
  return map
}
```

```typescript
      item = (await utils.reading.getById.fetch({ id: existing.id })) as ReadingItem
    } else {
      const article = await utils.reading.fetchWikipedia.fetch({ url })
      item = (await addWikipediaMutation.mutateAsync(article)) as ReadingItem
    }

    // Push to front of queue
    setCurrent(item)
    setQueue(prev => (current ? [current, ...prev] : prev))
    setLocalHiddenPassages([])
    utils.reading.getImportedWikipediaUrls.invalidate()
    toast.success(`Reading "${articleTitle}" next`, { id: toastId })
  } catch {
    toast.error(`Failed to open "${articleTitle}"`, { id: toastId })
  } finally {
    addingWikipediaUrls.current.delete(url)
  }
}
```
